---
title: Using Solvix with Microservices
description: Service-to-service communication with circuit breakers, health checks, and distributed tracing.
---

```ts
import { createClient } from "@adityadev13/solvix";

// Service A → Service B client
const serviceB = createClient({
  baseURL: "http://service-b.internal:3001",
  timeout: 3000,
  retry: { retries: 2 },
  circuitBreaker: {
    failureThreshold: 5,
    resetTimeout: 30000,
  },
  healthCheck: {
    enabled: true,
    endpoint: "/health",
    intervalMs: 10000,
  },
  tracing: { enabled: true },
  correlation: { enabled: true },
  logger: console,
});

// If Service B goes down:
// 1. Circuit opens after 5 failures
// 2. Health checker detects the outage
// 3. Requests fail fast instead of timing out
// 4. When B recovers, circuit closes automatically
const res = await serviceB.get("/api/data");
```
