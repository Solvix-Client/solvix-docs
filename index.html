Installation

npm install @adityadev13/solvix
client.ts
import { createClient } from "@adityadev13/solvix";

const client = createClient({
  baseURL: "https://api.example.com",
  retry: { retries: 3 },
});

const response = await client.get("/users");
console.log(response.data);
